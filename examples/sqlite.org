* Setup

Creates =example.db= next to this file, with =people=, =products=
and =orders= tables.  Run it again to reset the data.

* Read

#+begin_src datamacs :driver sqlite :file example.db
SELECT * FROM people
#+end_src

#+begin_src datamacs :driver sqlite :file example.db
SELECT p.name, pr.name AS product, o.quantity,
       round(o.quantity * pr.price, 2) AS total, o.ordered_at
FROM orders o
JOIN people p ON p.id = o.person_id
JOIN products pr ON pr.id = o.product_id
ORDER BY o.ordered_at DESC
LIMIT 10
#+end_src

#+begin_src datamacs :driver sqlite :file example.db
SELECT p.name, count(o.id) AS orders,
       round(coalesce(sum(o.quantity * pr.price), 0), 2) AS spent
FROM people p
LEFT JOIN orders o ON o.person_id = p.id
LEFT JOIN products pr ON pr.id = o.product_id
GROUP BY p.id
ORDER BY spent DESC
#+end_src

#+begin_src datamacs :driver sqlite :file example.db
SELECT pr.category, sum(o.quantity) AS items,
       round(sum(o.quantity * pr.price), 2) AS revenue
FROM orders o
JOIN products pr ON pr.id = o.product_id
GROUP BY pr.category
ORDER BY revenue DESC
#+end_src

#+begin_src datamacs :driver sqlite :file example.db
SELECT city, count(*) AS people, avg(age) AS avg_age
FROM people
GROUP BY city
ORDER BY people DESC
#+end_src

#+begin_src datamacs :driver sqlite :file example.db
UPDATE people SET age = age + 1 WHERE city = 'Tashkent'
#+end_src

* Read-only

Writes are refused on a =:readonly t= connection.

#+begin_src datamacs :driver sqlite :file example.db :readonly t
DELETE FROM people
#+end_src

Reads still work.

#+begin_src datamacs :driver sqlite :file example.db :readonly t
SELECT name, age FROM people WHERE age > 30
#+end_src
